cmake_minimum_required(VERSION 3.20)
project(lcb_ai VERSION 0.5.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(lcb_core STATIC src/conversation.c src/markdown.c vendor/cjson/cJSON.c)
target_include_directories(lcb_core PUBLIC src vendor/cjson)
target_compile_definitions(lcb_core PRIVATE CJSON_NESTING_LIMIT=32)
add_executable(core_test tests/core_test.c)
target_link_libraries(core_test PRIVATE lcb_core)
enable_testing()
add_test(NAME core COMMAND core_test)
add_executable(markdown_test tests/markdown_test.c)
target_link_libraries(markdown_test PRIVATE lcb_core)
add_test(NAME markdown COMMAND markdown_test)

set(lcb_targets lcb_core core_test markdown_test)
if(WIN32)
    list(APPEND lcb_targets lcb_transport lcb_models lcb_store lcb-ai lcb-cli engine_test http_test store_test library_test identity_test settings_test defaults_test context_live_test desktop_test)
endif()
foreach(target IN LISTS lcb_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/sdl;/GS;/guard:cf>")
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        if(target MATCHES "_test$")
            target_compile_options(${target} PRIVATE /UNDEBUG)
        endif()
        get_target_property(kind ${target} TYPE)
        if(kind STREQUAL "EXECUTABLE")
            target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
        endif()
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Werror;-fstack-protector-strong>")
        if(target MATCHES "_test$")
            target_compile_options(${target} PRIVATE -UNDEBUG)
        endif()
    endif()
endforeach()
