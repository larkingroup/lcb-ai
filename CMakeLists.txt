cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
project(lcb_ai VERSION 0.19.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(WIN32)
    # Windows 10 APIs such as BCryptHash; also the floor for Windows on ARM64.
    add_compile_definitions(_WIN32_WINNT=0x0A00 WINVER=0x0A00)
endif()

add_library(lcb_core STATIC src/conversation.c src/context.c src/markdown.c src/compatibility.c vendor/cjson/cJSON.c)
target_include_directories(lcb_core PUBLIC src vendor/cjson)
target_compile_definitions(lcb_core PRIVATE CJSON_NESTING_LIMIT=32)
if(WIN32)
    enable_language(RC)
    add_library(lcb_transport STATIC src/local.c src/stream_win.c src/context_win.c src/attachments_win.c)
    target_link_libraries(lcb_transport PUBLIC lcb_core winhttp bcrypt)
    add_library(lcb_models STATIC src/library_win.c src/model_defaults.c)
    target_link_libraries(lcb_models PUBLIC lcb_core)
    add_library(lcb_store STATIC src/store_win.c src/model_settings_win.c)
    target_link_libraries(lcb_store PUBLIC lcb_core lcb_models lcb_transport bcrypt)
    add_executable(lcb-ai WIN32 src/winmain.c src/classic_win.c src/engine_win.c src/editor_win.c src/setup_win.c src/app_paths.c assets/lcb.rc)
    target_include_directories(lcb-ai PRIVATE assets)
    target_link_libraries(lcb-ai PRIVATE lcb_transport lcb_store user32 gdi32 uxtheme dwmapi comdlg32 comctl32 shell32 ws2_32)
    add_executable(lcb-cli src/cli.c)
    target_link_libraries(lcb-cli PRIVATE lcb_transport)
endif()

set(lcb_targets lcb_core)
if(UNIX AND NOT APPLE)
    enable_language(CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    find_package(Threads REQUIRED)
    find_package(CURL REQUIRED)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    find_package(Fontconfig REQUIRED)
    option(LCB_FETCH_FLTK "Build pinned FLTK when no installed FLTK 1.4 exists" ON)
    find_package(FLTK 1.4 CONFIG QUIET)
    if(NOT FLTK_FOUND)
        if(NOT LCB_FETCH_FLTK)
            message(FATAL_ERROR "Install FLTK 1.4 development files or enable LCB_FETCH_FLTK")
        endif()
        include(FetchContent)
        set(FLTK_BUILD_TEST OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_FLUID OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_FORMS OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_GL OFF CACHE BOOL "" FORCE)
        set(FLTK_OPTION_PRINT_SUPPORT OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_HTML_DOCS OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_PDF_DOCS OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(fltk
            URL https://github.com/fltk/fltk/archive/refs/tags/release-1.4.5.tar.gz
            URL_HASH SHA256=7715e69ce081fa9ce6da48bb0dd3b07a4cf2cf937813814c04272f36fff593ea)
        FetchContent_MakeAvailable(fltk)
    endif()
    add_library(lcb_linux STATIC src/linux/native.cpp src/library_win.c src/model_defaults.c)
    target_compile_definitions(lcb_linux PRIVATE _GNU_SOURCE _POSIX_C_SOURCE=200809L)
    target_link_libraries(lcb_linux PUBLIC lcb_core CURL::libcurl OpenSSL::Crypto Threads::Threads)
    target_include_directories(lcb_linux PUBLIC src/linux)
    add_executable(lcb-ai src/linux/desktop.cpp)
    target_link_libraries(lcb-ai PRIVATE lcb_linux fltk::fltk fltk::images Fontconfig::Fontconfig)
    # The interface font ships beside the executable and loads privately.
    file(COPY vendor/wine-tahoma/tahoma.ttf vendor/wine-tahoma/tahomabd.ttf
         DESTINATION ${CMAKE_BINARY_DIR}/fonts)
    file(COPY assets/lcb-icon.svg assets/lcb-icon-small.svg DESTINATION ${CMAKE_BINARY_DIR}/assets)
    target_compile_definitions(lcb-ai PRIVATE LCB_VERSION="${PROJECT_VERSION}")
    add_executable(lcb-cli src/linux/cli.cpp)
    target_link_libraries(lcb-cli PRIVATE lcb_linux)
    set_target_properties(lcb-ai lcb-cli PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "")
    install(TARGETS lcb-ai lcb-cli RUNTIME DESTINATION bin COMPONENT lcb-ai)
    install(FILES assets/lcb-icon.svg DESTINATION share/icons/hicolor/scalable/apps RENAME lcb-ai.svg COMPONENT lcb-ai)
    install(FILES assets/lcb-icon-small.svg DESTINATION share/icons/hicolor/16x16/apps RENAME lcb-ai.svg COMPONENT lcb-ai)
    install(FILES assets/lcb-icon.svg assets/lcb-icon-small.svg DESTINATION share/lcb-ai COMPONENT lcb-ai)
    install(FILES src/linux/lcb-ai.desktop DESTINATION share/applications COMPONENT lcb-ai)
    install(FILES vendor/wine-tahoma/tahoma.ttf vendor/wine-tahoma/tahomabd.ttf
            DESTINATION share/lcb-ai/fonts COMPONENT lcb-ai)
    install(DIRECTORY vendor/wine-tahoma DESTINATION share/licenses/lcb-ai COMPONENT lcb-ai
            FILES_MATCHING PATTERN "LICENSE" PATTERN "*.md")
    install(FILES vendor/cjson/LICENSE DESTINATION share/licenses/lcb-ai RENAME cJSON-LICENSE COMPONENT lcb-ai)
    install(FILES vendor/fltk/LICENSE DESTINATION share/licenses/lcb-ai RENAME FLTK-LICENSE COMPONENT lcb-ai)
    install(FILES README.md DESTINATION share/doc/lcb-ai COMPONENT lcb-ai)
    list(APPEND lcb_targets lcb_linux lcb-ai lcb-cli)

    # cpack --config build-linux/CPackConfig.cmake writes a relocatable tarball,
    # plus an RPM or DEB when rpmbuild or dpkg-deb is installed.
    set(CPACK_PACKAGE_NAME lcb-ai)
    set(CPACK_PACKAGE_VENDOR "Larkin Group")
    set(CPACK_PACKAGE_CONTACT "https://github.com/larkingroup/lcb-ai")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Local AI workbench")
    set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/larkingroup/lcb-ai")
    set(CPACK_PACKAGE_FILE_NAME "lcb-ai-v${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}-linux-${CMAKE_SYSTEM_PROCESSOR}")
    set(CPACK_PACKAGE_INSTALL_DIRECTORY lcb-ai)
    # Package only the app, not the install rules of a fetched FLTK.
    set(CPACK_INSTALL_CMAKE_PROJECTS "${CMAKE_BINARY_DIR};lcb_ai;lcb-ai;/")
    set(CPACK_STRIP_FILES ON)
    set(CPACK_PACKAGE_CHECKSUM SHA256)
    set(CPACK_GENERATOR TGZ)
    find_program(LCB_RPMBUILD rpmbuild)
    if(LCB_RPMBUILD)
        list(APPEND CPACK_GENERATOR RPM)
    endif()
    find_program(LCB_DPKG_DEB dpkg-deb)
    if(LCB_DPKG_DEB)
        list(APPEND CPACK_GENERATOR DEB)
    endif()
    set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
    set(CPACK_RPM_PACKAGE_GROUP Applications/Productivity)
    set(CPACK_RPM_PACKAGE_AUTOREQ ON)
    set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share/applications /usr/share/icons
        /usr/share/icons/hicolor /usr/share/icons/hicolor/scalable /usr/share/icons/hicolor/scalable/apps
        /usr/share/icons/hicolor/16x16 /usr/share/icons/hicolor/16x16/apps /usr/share/licenses)
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_DEBIAN_PACKAGE_SECTION utils)
    find_program(LCB_DPKG_SHLIBDEPS dpkg-shlibdeps)
    if(LCB_DPKG_SHLIBDEPS)
        set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    else()
        set(CPACK_DEBIAN_PACKAGE_DEPENDS "libcurl4, libssl3, libfontconfig1, libxft2, libcairo2, libpango-1.0-0, libwayland-client0, libxkbcommon0")
    endif()
    include(CPack)
endif()
if(WIN32)
    list(APPEND lcb_targets lcb_transport lcb_models lcb_store lcb-ai lcb-cli)
endif()
foreach(target IN LISTS lcb_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C>:/W4;/WX;/sdl;/GS;/guard:cf>")
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        get_target_property(kind ${target} TYPE)
        if(kind STREQUAL "EXECUTABLE")
            target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
        endif()
    else()
        target_compile_options(${target} PRIVATE "$<$<COMPILE_LANGUAGE:C,CXX>:-Wall;-Wextra;-Werror;-fstack-protector-strong>")
    endif()
endforeach()
