cmake_minimum_required(VERSION 3.20)
project(lti_ai VERSION 0.3.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(lti_core STATIC src/conversation.c vendor/cjson/cJSON.c)
target_include_directories(lti_core PUBLIC src vendor/cjson)
target_compile_definitions(lti_core PRIVATE CJSON_NESTING_LIMIT=32)
if(WIN32)
    add_library(lti_transport STATIC src/local.c)
    target_link_libraries(lti_transport PUBLIC lti_core winhttp)
    add_library(lti_store STATIC src/store_win.c)
    target_link_libraries(lti_store PUBLIC lti_core bcrypt)
    add_executable(lti-ai WIN32 src/winmain.c src/classic_win.c src/engine_win.c src/editor_win.c src/library_win.c)
    target_link_libraries(lti-ai PRIVATE lti_transport lti_store user32 gdi32 uxtheme dwmapi comdlg32 comctl32 ws2_32)
    add_executable(lti-cli src/cli.c)
    target_link_libraries(lti-cli PRIVATE lti_transport)
endif()
set(lti_targets lti_core)
if(WIN32)
    list(APPEND lti_targets lti_transport lti_store lti-ai lti-cli)
endif()
foreach(target IN LISTS lti_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /sdl /GS /guard:cf)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        get_target_property(kind ${target} TYPE)
        if(kind STREQUAL "EXECUTABLE")
            target_link_options(${target} PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
    endif()
endforeach()
